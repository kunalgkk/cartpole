import gym

env = gym.make("CartPole-v1")
episodes = 50

total_scores = []

for episode in range(episodes):
    state = env.reset()[0]
    total_reward = 0

    for t in range(500):
        action = env.action_space.sample()
        next_state, reward, done, _, _ = env.step(action)
        total_reward += reward
        state = next_state
        if done:
            break

    total_scores.append(total_reward)

print("Average Random Policy Score:", sum(total_scores)/episodes)
